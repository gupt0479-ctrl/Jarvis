```python
GREENHOUSE_URL = "https://boards-api.greenhouse.io/v1/boards/{token}/jobs"
LEVER_URL = "https://api.lever.co/v0/postings/{token}?mode=json"
ASHBY_URL = "https://api.ashbyhq.com/posting-api/job-board/{token}"

def fetch_greenhouse(token: str, http_get=None) -> list:
    resp = (http_get or requests.get)(GREENHOUSE_URL.format(token=token), timeout=TIMEOUT)
    resp.raise_for_status()
    return [normalize_greenhouse(raw, token) for raw in resp.json()["jobs"]]
```
