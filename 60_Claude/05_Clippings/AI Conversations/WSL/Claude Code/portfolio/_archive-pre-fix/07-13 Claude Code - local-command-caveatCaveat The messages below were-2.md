"use client";

import { buildSmoothPath } from "@/lib/svg-chart";

const VIEW_W = 120;
const VIEW_H = 40;
const PAD = 4;

// Decorative "trending up" fallback shape — used when there isn't enough
// real structured data (fewer than 2 skills/projects) to plot.
const FALLBACK_POINTS: [number, number][] = [
  [0, 0.8],
  [1, 0.65],
  [2, 0.8],
  [3, 0.45],
  [4, 0.15],
];

function toSvgPoints(values: [number, number][]): [number, number][] {
  const xs = values.map((p) => p[0]);
  const ys = values.map((p) => p[1]);
  const minX = Math.min(...xs);
  const maxX = Math.max(...xs);
  const minY = Math.min(...ys);
  const maxY = Math.max(...ys);
  const xRange = maxX - minX || 1;
  const yRange = maxY - minY || 1;
  return values.map(([x, y]) => [
    PAD + ((x - minX) / xRange) * (VIEW_W - PAD * 2),
    PAD + (VIEW_H - PAD * 2) - ((y - minY) / yRange) * (VIEW_H - PAD * 2),
  ]);
}

export function TelemetryDetail({
  points,
  summary,
  label,
}: {
  points: [number, number][] | null;
  summary?: string | null;
  label: string;
}) {
  const source = points && points.length >= 2 ? points : FALLBACK_POINTS;
  const path = buildSmoothPath(toSvgPoints(source));

  return (
    <div className="mt-3 pt-3 border-t border-white/[0.06]">
      <svg
        viewBox={`0 0 ${VIEW_W} ${VIEW_H}`}
        className="h-10 w-full"
        role="img"
        aria-label={`Trend graph for ${label}`}
      >
        <path
          d={path}
          fill="none"
          stroke="rgba(167,139,250,0.7)"
          strokeWidth={2}
          strokeLinecap="round"
        />
      </svg>
      {summary && (
