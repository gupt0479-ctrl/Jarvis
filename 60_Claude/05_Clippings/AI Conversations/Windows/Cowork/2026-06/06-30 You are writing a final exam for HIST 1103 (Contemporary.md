const SZ    = 24;           // 12 pt in half-points

function body(texts, fnId) {
  // texts is an array of {text, bold?, italic?, fnAfter?} or plain string
  // fnId: optional – if provided, a FootnoteReferenceRun is appended
  const runs = [];
  if (Array.isArray(texts)) {
    for (const t of texts) {
      if (typeof t === 'string') {
        runs.push(new TextRun({ text: t, font: TNR, size: SZ }));
      } else {
        runs.push(new TextRun({ text: t.text, font: TNR, size: SZ, bold: t.bold, italics: t.italic }));
      }
    }
  } else {
    runs.push(new TextRun({ text: texts, font: TNR, size: SZ }));
  }
  if (fnId !== undefined) {
    runs.push(new FootnoteReferenceRun(fnId));
  }
  return new Paragraph({
    spacing: { ...DBL, before: 0, after: 0 },
    indent: { firstLine: 720 },          // 0.5 in first-line indent
    children: runs
  });
}

function bodyInline(segments) {
  // segments: array of {text, bold?, italic?, fn?}  — fn inserts footnote ref after text
  const runs = [];
  for (const seg of segments) {
    if (typeof seg === 'string') {

    } else {

      if (seg.fn !== undefined) runs.push(new FootnoteReferenceRun(seg.fn));
    }
  }
  return new Paragraph({

    indent: { firstLine: 720 },

  });

  return new Paragraph({
    spacing: { line: 480, lineRule: 'auto', before: 240, after: 0 },
    children: [new TextRun({ text, font: TNR, size: SZ, bold: true })]
  });
}

function subheading(text) {
  return new Paragraph({
    spacing: { line: 480, lineRule: 'auto', before: 120, after: 0 },
    children: [new TextRun({ text, font: TNR, size: SZ, bold: true, underline: {} })]
  });
}

function empty() {
  return new Paragraph({ spacing: { line: 480, lineRule: 'auto' }, children: [] });
}

function bibEntry(text) {
  return new Paragraph({
    spacing: { line: 240, lineRule: 'auto', before: 0, after: 120 },
    indent: { left: 720, hanging: 720 },
    children: [new TextRun({ text, font: TNR, size: SZ })]
  });
}

function fn(id, textSegments) {
  // textSegments: array of {text, italic?} or plain strings
  const runs = [];
  for (const seg of textSegments) {
    if (typeof seg === 'string') {
      runs.push(new TextRun({ text: seg, font: TNR, size: SZ_SM }));
    } else {
      runs.push(new TextRun({ text: seg.text, font: TNR, size: SZ_SM, italics: seg.italic }));
    }
  }
  return { [id]: { children: [new Paragraph({ spacing: SGL, children: runs })] } };
}
